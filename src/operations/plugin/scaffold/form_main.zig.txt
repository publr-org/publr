//! One set of settings, edited on a page of its own.
//!
//! Read them with `__name__ read`, change them with `__name__ save`. The page in the admin is
//! a form drawn from `save`'s input, filled from what `read` answers.
const std = @import("std");
const publr = @import("publr");

const manifest: publr.Manifest = .{
    .name = "__name__",
    .version = "0.1.0",
};

/// __Title__ settings
///
/// The plugin's one set of settings.
pub const Settings = struct {
    /// Title
    ///
    /// The site's name.
    title: []const u8,
    /// Description
    ///
    /// Shown under the title.
    description: []const u8 = "",
};

pub const Stored = publr.record.define(Settings, .{
    .name = "__name__.settings",
    .kind = .settings,
    // Edited on the plugin's own page; no second entry in the menu.
    .menu = .none,
    // The record checks its own data on every write, whoever writes; the rules also show
    // beside each field's help.
    .fields = .{
        .title = .{
            .validation = .{
                .min_len = 1,
                .max_len = 160,
            },
        },
        .description = .{
            .kind = "text",
            .validation = .{
                .max_len = 2000,
            },
        },
    },
});

const the_settings = publr.query.prepare(
    \\*[_type == "__name__.settings"][0]{ "id": _id, title, description }
, .{});

const Found = struct {
    id: []const u8,
    title: []const u8,
    description: []const u8 = "",
};

pub const ReadIn = struct {};

pub const ReadOut = struct {
    /// The site's name
    title: []const u8,
    /// Shown under the title
    description: []const u8 = "",
};

/// The settings as they are.
///
/// Administrators, and callers granted `__name__.read`. Before the first save, an empty
/// title.
///
/// Example:
///   in: {}
///   out: { "title": "My site", "description": "" }
pub const Read = publr.operation.define(ReadIn, ReadOut, .{
    .name = "read",
    .kind = .read,
    .run = read,
});

fn read(ctx: *publr.Ctx, _: ReadIn) publr.Error!ReadOut {
    std.debug.assert(ctx.now().ms >= 0);

    const found = try the_settings.execute(ctx, ?Found, .{}) orelse return .{ .title = "" };

    return .{
        .title = found.title,
        .description = found.description,
    };
}

pub const SaveIn = struct {
    /// The site's name
    title: []const u8,
    /// Shown under the title
    description: []const u8 = "",
};

pub const SaveOut = struct {
    /// The settings' id
    id: []const u8,
};

/// Changes the settings.
///
/// Administrators, and callers granted `__name__.save`.
///
/// Example:
///   in: { "title": "My site", "description": "" }
///   out: { "id": "0a1b2c3d4e5f60718293a4b5" }
pub const Save = publr.operation.define(SaveIn, SaveOut, .{
    .name = "save",
    .kind = .write,
    .run = save,
});

fn save(ctx: *publr.Ctx, in: SaveIn) publr.Error!SaveOut {
    std.debug.assert(in.title.len <= 1 << 16);

    const value: Settings = .{
        .title = in.title,
        .description = in.description,
    };
    const found = try the_settings.execute(ctx, ?Found, .{}) orelse {
        return .{ .id = try Stored.create(ctx, value) };
    };

    _ = try Stored.save(ctx, found.id, value, .{});

    return .{
        .id = found.id,
    };
}

const settings: publr.plugin.Page = .{
    .path = "settings",
    .label = "__Title__",
    .icon = "settings",
    .form = .{
        .read = Read,
        .save = Save,
    },
};

pub const plugin = publr.plugin.define(manifest, .{
    .records = &.{Stored},
    .operations = &.{ Read, Save },
    .pages = &.{settings},
});
