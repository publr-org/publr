//! Entries people add and remove, on a page of its own.
//!
//! List the entries with `__name__ list`, add one with `__name__ add`, remove one with
//! `__name__ remove`. The page in the admin shows them all.
const std = @import("std");
const publr = @import("publr");

const manifest: publr.Manifest = .{
    .name = "__name__",
    .version = "0.1.0",
};

/// __Title__ entry
///
/// One entry people add on the plugin's page.
pub const Entry = struct {
    /// Title
    title: []const u8,
};

pub const Entries = publr.record.define(Entry, .{
    .name = "__name__.entry",
    // Checked on every write, whoever writes.
    .fields = .{
        .title = .{
            .validation = .{
                .min_len = 1,
                .max_len = 160,
            },
        },
    },
});

const every_entry = publr.query.prepare(
    \\*[_type == "__name__.entry"][0...200]{ "id": _id, title }
, .{});

pub const Listed = struct {
    /// The entry's id
    id: []const u8,
    /// Its title
    title: []const u8,
};

pub const ListIn = struct {};

pub const ListOut = struct {
    /// Every entry, newest last
    entries: []const Listed,
};

/// Every entry, newest last.
///
/// Administrators, and callers granted `__name__.list`. Changes nothing.
///
/// Example:
///   in: {}
///   out: { "entries": [{ "id": "0a1b2c3d4e5f60718293a4b5", "title": "First" }] }
pub const List = publr.operation.define(ListIn, ListOut, .{
    .name = "list",
    .kind = .read,
    .run = list,
});

fn list(ctx: *publr.Ctx, _: ListIn) publr.Error!ListOut {
    std.debug.assert(ctx.now().ms >= 0);

    return .{
        .entries = try every_entry.execute(ctx, []const Listed, .{}),
    };
}

pub const AddIn = struct {
    /// The entry's title
    title: []const u8,
};

pub const AddOut = struct {
    /// The new entry's id
    id: []const u8,
};

/// Adds an entry.
///
/// Administrators, and callers granted `__name__.add`.
///
/// Example:
///   in: { "title": "First" }
///   out: { "id": "0a1b2c3d4e5f60718293a4b5" }
pub const Add = publr.operation.define(AddIn, AddOut, .{
    .name = "add",
    .kind = .write,
    .run = add,
});

fn add(ctx: *publr.Ctx, in: AddIn) publr.Error!AddOut {
    std.debug.assert(in.title.len <= 1 << 16);

    return .{
        .id = try Entries.create(ctx, .{ .title = in.title }),
    };
}

pub const RemoveIn = struct {
    /// The entry to remove
    id: []const u8,
};

pub const RemoveOut = struct {
    /// Whether it was removed
    removed: bool,
};

/// Removes an entry.
///
/// Administrators, and callers granted `__name__.remove`. Fails as not found for an id
/// that is not an entry.
///
/// Example:
///   in: { "id": "0a1b2c3d4e5f60718293a4b5" }
///   out: { "removed": true }
pub const Remove = publr.operation.define(RemoveIn, RemoveOut, .{
    .name = "remove",
    .kind = .write,
    .run = remove,
});

fn remove(ctx: *publr.Ctx, in: RemoveIn) publr.Error!RemoveOut {
    std.debug.assert(in.id.len > 0);

    _ = try Entries.get(ctx, in.id);
    _ = try ctx.call(publr.operations.record.Delete, .{ .id = in.id });

    return .{
        .removed = true,
    };
}

pub const PageIn = struct {
    /// The address's query string
    query: []const u8 = "",
    /// The path the project is served under
    base: []const u8 = "",
};

pub const PageOut = struct {
    /// The page, drawn
    html: []const u8,
};

/// Draws the plugin's page in the admin.
///
/// Administrators, and callers granted `__name__.page`. Changes nothing.
///
/// Example:
///   in: {}
///   out: { "html": "<!doctype html>" }
pub const Page = publr.operation.define(PageIn, PageOut, .{
    .name = "page",
    .kind = .read,
    .run = page,
});

fn page(ctx: *publr.Ctx, in: PageIn) publr.Error!PageOut {
    std.debug.assert(in.query.len <= 4096);

    const listed = try list(ctx, .{});

    // Compiled in, the admin's layouts draw it; installed, until views reach the frame.
    if (comptime @import("builtin").os.tag != .freestanding) {
        return .{ .html = try drawn(ctx, in.base, listed.entries) };
    }

    return .{ .html = "<p>This page needs the plugin compiled in for now.</p>" };
}

/// The page, drawn by the admin's layouts: each entry as the list shows it.
fn drawn(ctx: *publr.Ctx, base: []const u8, listed: []const Listed) publr.Error![]const u8 {
    const View = publr.admin.views.__Name__Page;
    const Item = std.meta.Child(@FieldType(View.Props, "items"));
    const arena = ctx.arena();
    const items = arena.alloc(Item, listed.len) catch return error.OutOfMemory;

    for (listed, items) |entry, *item| {
        item.* = .{ .id = entry.id, .title = entry.title, .type_label = "Entry", .status = "", .tone = "", .href = "" };
    }

    const request = publr.admin.chrome.in_plugin_frame(ctx, base) catch return error.OutOfMemory;

    return publr.admin.render.document(arena, request, View, .{ .items = items }) catch error.OutOfMemory;
}

const entries: publr.plugin.Page = .{
    .path = "entries",
    .label = "__Title__",
    .icon = "list-unordered",
    .operation = Page,
};

pub const plugin = publr.plugin.define(manifest, .{
    .records = &.{Entries},
    .operations = &.{ List, Add, Remove, Page },
    .pages = &.{entries},
});
